\begin{denotation}[3.5cm]

\item[$S$]{
  臺大校園周遭站點集合
}

\item[$C_i$]{
  站點 $i$ 之車柱總容量（Capacity）
}

\item[$N$]{
  臺大校園全區可用單車總數上限
}

\item[$V$]{
  營運調度卡車集合
}

\item[$\hat{C}_v$]{
  調度卡車 $v$ 之最大載運容量
}

\item[$\tau$]{
  1 分鐘微觀快照時間索引（$\tau \in \{1, \dots, 1440\}$）
}

\item[$t$]{
  宏觀決策規劃時段索引（$t \in T$）
}

\item[$B^{\text{eff}}_{i,\tau}$]{
  時點 $\tau$ 站點 $i$ 排除故障車後之真實有效可用車數
}

\item[$\text{Exposure}^{\pm}_{i,s}$]{
  時段 $s$ 內有車可借（$+$）或有位可還（$-$）之未截斷暴露時間
}

\item[$\hat{\lambda}_{i,t}, \hat{\mu}_{i,t}$]{
  經截斷校正後站點 $i$ 於時段 $t$ 之真實潛在借車率與還車率
}

\item[$P_t(i,j)$]{
  時段 $t$ 由站點 $i$ 借出並於站點 $j$ 歸還之時變馬可夫轉移機率
}

\item[$p_i(x, \sigma, t)$]{
  期初在庫為 $x$ 下，時刻 $t$ 站點庫存為 $\sigma$ 之暫態機率分佈
}

\item[$U_{i,t}(x)$]{
  站點 $i$ 於時段 $t$ 之預期使用者不滿意度凸函數（UDF）
}

\item[$\ell^*_{i,t}$]{
  站點 $i$ 於時段 $t$ 之單站最適目標庫存水位
}

\item[{$[\underline{\ell}_{i,t}, \bar{\ell}_{i,t}]$}]{
  站點 $i$ 於時段 $t$ 之 Type-2 服務水準安全庫存緩衝區間
}

\item[$d^t_i, \hat{d}^t_v$]{
  時段 $t$ 站點 $i$ 在庫車數與調度卡車 $v$ 車上載車量
}

\item[$r^{+, t}_{i,v}, r^{-, t}_{i,v}$]{
  時段 $t$ 卡車 $v$ 於站點 $i$ 之移出裝載量與補入卸載量
}

\item[$e^{-, t}_i, e^{+, t}_i$]{
  站點庫存跌破安全下界與超出安全上界之軟緩衝懲罰偏差量
}

\item[$\text{Score}$]{
  整合使用者遺憾、搬運作業與行駛里程之多視角綜合評分
}

\end{denotation}
